```
# 1. 安装 Node.js（阿里镜像） 
brew install node node --version # 验证 Node 可用 
npm --version # 验证 npm 可用 
# 2. 安装 Python 3.13（阿里镜像同步完成，无限制） 
brew install python@3.13 python3.13 --version # 输出 3.13.3 
# 3. 配置 pip 绕过 PEP 668 限制（仅用户级生效，安全） 
mkdir -p ~/.config/pip echo -e "[global]\nbreak-system-packages = true\nuser = true" > ~/.config/pip/pip.conf 
# 4. 安装 Poetry（无 SSL/权限报错） python3.13 -m pip install poetry --upgrade poetry --version # 输出最新版 
# 5. 验证所有工具路径（确保在 brew 目录下） 
which node # /opt/homebrew/bin/node 
which python3.13 # /opt/homebrew/bin/python3.13 
which poetry # /Users/nian/.local/bin/poetry
```

# 系统设置

## 清除软件隔离属性

老方法：

```shell
sudo spctl --master-disablesudo 
sudo xattr -dr com.apple.quarantine  // 从应用程序文件夹把程序拖放到这里
```

新方法：

### 第一步：创建通用命令（一次性配置）

打开终端，编辑 `~/.zshrc`（Mac 默认终端配置文件）：

```shell
nano ~/.zshrc
# 或者
open -e ~/.zshrc
# 或者
vim ~/.zshrc
```

vim 模式按 `i` 进入编辑模式，粘贴以下代码（通用清除隔离属性的函数）：

```bash
    # 通用清除 macOS 隔离属性命令（拖文件/应用到终端即可）
    function unquarantine() {
        # 检查是否传入了文件/应用路径
        if [ -z "$1" ]; then
            echo "用法：unquarantine 拖文件/应用到这里"
            return 1
        fi
        # 适配老版本 macOS（无 -r 参数），直接清空所有扩展属性
        sudo xattr -c "$1"
        # 提示执行成功
        echo "✅ 已清除「$1」的隔离属性，现在可以打开了！"
    }
# 给函数加短别名（核心：把 unquarantine 缩写成 uq）
alias uq='unquarantine'
    ```
按 `esc` → 输入 `:wq` 保存并退出 vim；
让配置生效：
```bash
    source ~/.zshrc
```

### 第二步：以后用的话，只需 2 步（无脑操作）

1. 终端输入 `uq` （注意末尾加一个空格，别漏！）；
2. 把要打开的软件（比如 `/Applications/XXX.app`）从访达拖到终端里 → 回车；
3. 输入电脑密码（终端输入密码时不显示，正常输就行）→ 回车，搞定！

#### 示例操作（复制就能看效果）

```bash
# 输入命令+空格 → 拖应用 → 最终长这样
uq /Applications/Kiro\ Account\ Manager.app
```

执行后终端会输出 `✅ 已清除...`，直接打开软件就行。
